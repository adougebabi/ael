package main

import (
	"ael/internal/core"
	"context"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"testing"
	"time"
)

// Run through a temporary Go overlay. The fixture's fake git cannot publish.
func TestV212ApprovedCheckCannotLaunchIndirectPublish(t *testing.T) {
	makePath, err := exec.LookPath("make")
	if err != nil {
		t.Fatal(err)
	}
	for _, kind := range []string{"path", "absolute"} {
		t.Run(kind, func(t *testing.T) {
			workspace, bin := t.TempDir(), t.TempDir()
			fakeGit := filepath.Join(bin, "git")
			if err := os.WriteFile(fakeGit, []byte("#!/bin/sh\nprintf attempted > publish-marker\nexit 42\n"), 0700); err != nil {
				t.Fatal(err)
			}
			program := "git"
			if kind == "absolute" {
				program = "'" + strings.ReplaceAll(fakeGit, "'", "'\\''") + "'"
			}
			if err := os.WriteFile(filepath.Join(workspace, "Makefile"), []byte(".PHONY: test\ntest:\n\t"+program+" push origin fixture\n"), 0600); err != nil {
				t.Fatal(err)
			}
			t.Setenv("PATH", bin+string(os.PathListSeparator)+os.Getenv("PATH"))
			command := []string{makePath, "test"}
			check := core.Check{ID: "unit", Kind: "functional", Method: "command", Command: command}
			task := core.Task{Definition: core.Definition{ID: "fixture", Checks: []core.Check{check}}, Step: "verify"}
			cfg := remoteRunnerConfig{Workspace: workspace, Checks: []remoteApprovedCheck{{ID: "unit", Command: command}}}
			ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
			defer cancel()
			result, err := runRemoteProcess(ctx, cfg, task, core.Attempt{ID: "attempt", Unit: "check:unit"}, nil, t.TempDir(), t.TempDir())
			if err != nil {
				t.Fatal(err)
			}
			if _, err = os.Stat(filepath.Join(workspace, "publish-marker")); err == nil {
				t.Fatalf("an authorized check launched the unapproved publish command; result=%s", result.Result.Status)
			} else if !os.IsNotExist(err) {
				t.Fatal(err)
			}
		})
	}
}
