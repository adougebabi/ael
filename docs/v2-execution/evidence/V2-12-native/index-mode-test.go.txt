package remote

import (
    "strings"
    "testing"
)

func TestFixtureCandidateIndexModeObservation(t *testing.T) {
    source, _ := candidateRepos(t)
    first, err := CapturePortableCandidate(source, "fixture-repo")
    if err != nil { t.Fatal(err) }
    fixtureGit(t, source, "update-index", "--chmod=+x", "--", "app.txt")
    second, err := CapturePortableCandidate(source, "fixture-repo")
    if err != nil { t.Fatal(err) }
    if first.ID != second.ID { t.Fatalf("fixture no longer reproduces: %s %s", first.ID, second.ID) }
    raw, err := portableGit(source, "diff", "--cached", "--summary", "--", "app.txt")
    if err != nil || !strings.Contains(string(raw), "mode change 100644 => 100755") { t.Fatalf("missing staged mode difference: %q %v", raw, err) }
    t.Log("candidate ID unchanged although staged mode changed 100644 => 100755")
}
